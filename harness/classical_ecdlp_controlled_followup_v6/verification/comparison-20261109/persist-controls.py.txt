from pathlib import Path
import hashlib,json,os,subprocess,tarfile
root=Path('/tmp/ecd-develop-20261109');repo=root/'repository'
py='/workspace/classical-ecdlp-endpoint-reuse/runtime-20261028/bin/python';aws='/workspace/classical-ecdlp-endpoint-reuse/runtime-20261028/bin/aws'
identity=json.loads(subprocess.check_output([aws,'sts','get-caller-identity'],text=True));assert identity['Account']=='104186811093'
meta=json.loads((root/'package-metadata.json').read_text());archive=Path(meta['archive']);rel=archive.relative_to(repo).as_posix()
env=os.environ.copy();env.update(ARTIFACTS_AWS=aws,ARTIFACTS_CACHE=str(root/'artifact-cache'))
cmd=[py,'-B','tools/artifacts/artifacts.py','push',rel]

proof={'schema':'ecd-software-controls.storage-proof/v1','source_commit':meta['source_commit'],
