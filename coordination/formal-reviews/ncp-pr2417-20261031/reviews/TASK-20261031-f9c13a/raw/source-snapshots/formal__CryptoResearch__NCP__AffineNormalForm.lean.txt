/-
Source: research/THM_COMMUTATOR_KERNEL1.md, Lemma 1.
Target: formal/targets/ncp-affine-normal-form.yaml.

`ZMod n` is the cyclic group in coordinates with generator `1`. Letters are
applied from left to right, as in the archived instrument. A translation's
signed count uses the number of negation letters to its RIGHT, not its left.
This theorem is about the group action; it makes no search-cost claim.
-/
import Mathlib.Data.ZMod.Basic
import Mathlib.Algebra.BigOperators.Group.Finset.Basic
import Mathlib.Algebra.Module.BigOperators
import Mathlib.Tactic.Abel
import Mathlib.Tactic.Ring

namespace CryptoResearch.NCP

inductive Letter (B : Nat) where
  | trans : Fin B → Letter B
  | neg : Letter B
deriving DecidableEq

def negCount {B : Nat} : List (Letter B) → Nat
  | [] => 0
  | .trans _ :: w => negCount w
  | .neg :: w => negCount w + 1

/-- `signedCount w i` is the sum of one sign per occurrence of `trans i`,
where the sign is `(-1)` to the number of negations after that occurrence. -/
def signedCount {B : Nat} : List (Letter B) → Fin B → Int
  | [], _ => 0
  | .trans j :: w, i => signedCount w i + if i = j then (-1 : Int) ^ negCount w else 0
  | .neg :: w, i => signedCount w i

def shadowSign {B : Nat} (w : List (Letter B)) : Int := (-1 : Int) ^ negCount w

theorem shadowSign_eq_one_or_neg_one {B : Nat} (w : List (Letter B)) :
    shadowSign w = 1 ∨ shadowSign w = -1 := by
  simpa [shadowSign] using neg_one_pow_eq_or ℤ (negCount w)

private theorem shadowSign_cons_neg {B : Nat} (w : List (Letter B)) :
    shadowSign (.neg :: w) = -shadowSign w := by
  simp [shadowSign, negCount, pow_succ]

def actLetter {B n : Nat} (s : Fin B → ZMod n) : Letter B → ZMod n → ZMod n
  | .trans i, x => x + s i
  | .neg, x => -x

/-- Leftmost letter acts first. -/
def wordAction {B n : Nat} (s : Fin B → ZMod n)
    (w : List (Letter B)) (x : ZMod n) : ZMod n :=
  w.foldl (fun y l => actLetter s l y) x

private theorem bias_trans {B n : Nat} (s : Fin B → ZMod n)
    (j : Fin B) (w : List (Letter B)) :
    (∑ i : Fin B, ((signedCount (.trans j :: w) i : Int) : ZMod n) * s i) =
      (∑ i : Fin B, ((signedCount w i : Int) : ZMod n) * s i) +
        ((shadowSign w : Int) : ZMod n) * s j := by
  simp [signedCount, shadowSign, add_mul, Finset.sum_add_distrib]

/-- Every word acts by the affine map encoded by its parity and positional
signed counts. The statement holds for every `n`; in particular it holds for
the positive-order cyclic group of the frozen task. -/
theorem word_action_affine_normal_form {B n : Nat} (s : Fin B → ZMod n)
    (w : List (Letter B)) (x : ZMod n) :
    wordAction s w x =
      ((shadowSign w : Int) : ZMod n) * x +
        ∑ i : Fin B, ((signedCount w i : Int) : ZMod n) * s i := by
  induction w generalizing x with
  | nil => simp [wordAction, shadowSign, signedCount, negCount]
  | cons l w ih =>
      cases l with
      | trans j =>
          change wordAction s w (x + s j) = _
          rw [ih, bias_trans]
          simp only [shadowSign, negCount]
          ring
      | neg =>
          change wordAction s w (-x) = _
          rw [ih]
          simp only [shadowSign, negCount, signedCount]
          simp only [pow_succ, Int.cast_mul, Int.cast_neg, Int.cast_one]
          ring

/-- The orbit value at the chosen generator is determined by the shadow. -/
theorem orbit_value_eq_shadow {B n : Nat} (s : Fin B → ZMod n)
    (w : List (Letter B)) :
    wordAction s w 1 =
      ((shadowSign w : Int) : ZMod n) +
        ∑ i : Fin B, ((signedCount w i : Int) : ZMod n) * s i := by
  simpa using word_action_affine_normal_form s w (1 : ZMod n)

/-- Equal shadows induce identical actions, not merely equal values at `1`. -/
theorem same_shadow_same_action {B n : Nat} (s : Fin B → ZMod n)
    (w v : List (Letter B))
    (hsign : shadowSign w = shadowSign v)
    (hcount : ∀ i : Fin B, signedCount w i = signedCount v i) :
    ∀ x : ZMod n, wordAction s w x = wordAction s v x := by
  intro x
  rw [word_action_affine_normal_form, word_action_affine_normal_form, hsign]
  congr 1
  apply Finset.sum_congr rfl
  intro i _
  rw [hcount i]

/-- The same normal form on an arbitrary additive group. Setting
`T i = sᵢ • P` gives the action on a cyclic subgroup generated by `P`;
the coordinate theorem above uses `G = ZMod n` and `P = 1`. -/
def groupLetterAction {B : Nat} {G : Type*} [AddCommGroup G]
    (T : Fin B → G) : Letter B → G → G
  | .trans i, x => x + T i
  | .neg, x => -x

def groupWordAction {B : Nat} {G : Type*} [AddCommGroup G]
    (T : Fin B → G) (w : List (Letter B)) (x : G) : G :=
  w.foldl (fun y l => groupLetterAction T l y) x

private theorem group_bias_trans {B : Nat} {G : Type*} [AddCommGroup G]
    (T : Fin B → G) (j : Fin B) (w : List (Letter B)) :
    (∑ i : Fin B, signedCount (.trans j :: w) i • T i) =
      (∑ i : Fin B, signedCount w i • T i) + shadowSign w • T j := by
  simp [signedCount, shadowSign, add_zsmul, Finset.sum_add_distrib]

theorem group_word_action_affine_normal_form {B : Nat} {G : Type*} [AddCommGroup G]
    (T : Fin B → G) (w : List (Letter B)) (x : G) :
    groupWordAction T w x =
      shadowSign w • x + ∑ i : Fin B, signedCount w i • T i := by
  induction w generalizing x with
  | nil => simp [groupWordAction, shadowSign, signedCount, negCount]
  | cons l w ih =>
      cases l with
      | trans j =>
          change groupWordAction T w (x + T j) = _
          rw [ih, group_bias_trans]
          simp only [shadowSign, negCount]
          simp only [zsmul_add]
          abel
      | neg =>
          change groupWordAction T w (-x) = _
          rw [ih]
          rw [shadowSign_cons_neg]
          simp only [signedCount, zsmul_neg, neg_zsmul]

/-- If each translation is an integer multiple of a generator `P`, the bias
is the signed scalar sum times `P`. No order assumption is needed; taking
`n • P = 0` makes the scalar sum an element of `ZMod n`. -/
theorem cyclic_word_action_affine_normal_form {B : Nat} {G : Type*} [AddCommGroup G]
    (P : G) (s : Fin B → Int) (w : List (Letter B)) (x : G) :
    groupWordAction (fun i => s i • P) w x =
      shadowSign w • x + (∑ i : Fin B, signedCount w i * s i) • P := by
  rw [group_word_action_affine_normal_form]
  congr 1
  rw [Finset.sum_smul]
  apply Finset.sum_congr rfl
  intro i _
  rw [mul_zsmul]

theorem group_same_shadow_same_action {B : Nat} {G : Type*} [AddCommGroup G]
    (T : Fin B → G) (w v : List (Letter B))
    (hsign : shadowSign w = shadowSign v)
    (hcount : ∀ i : Fin B, signedCount w i = signedCount v i) :
    ∀ x : G, groupWordAction T w x = groupWordAction T v x := by
  intro x
  rw [group_word_action_affine_normal_form, group_word_action_affine_normal_form, hsign]
  congr 1
  apply Finset.sum_congr rfl
  intro i _
  rw [hcount i]
